cmake_minimum_required(VERSION 3.24)
project(wvd_maa_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MAA_SDK "" CACHE PATH "Locked MaaFramework SDK")
add_executable(wvd_maa_probe core/main.cpp adapters/fake/fake.cpp adapters/windows/platform.cpp adapters/windows/device.cpp)
target_include_directories(wvd_maa_probe PRIVATE "${MAA_SDK}/include" core)
target_link_libraries(wvd_maa_probe PRIVATE "${MAA_SDK}/lib/MaaFramework.lib" psapi)
target_compile_options(wvd_maa_probe PRIVATE /utf-8 /W4 /EHsc)
target_compile_definitions(wvd_maa_probe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_options(wvd_maa_probe PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/adapters/windows/app.manifest")
